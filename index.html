email-generator.py

# This is my email generator!
# Lines starting with # are notes (computer ignores them)

# --- MY CANDIDATE LIST ---
candidates = [
    {"name": "Sarah", "role": "Software Engineer"},
    {"name": "James", "role": "Product Manager"},
    {"name": "Lisa", "role": "Data Scientist"},
]

# --- MY EMAIL TEMPLATE ---
template = """Hi {name},

I saw your profile and loved your experience as a {role}.

We have a role that's a great fit for your background.
Would you be open to a quick 15-min chat this week?

Best,
Clarice"""

# --- GENERATE EMAILS ---
for person in candidates:
    print("─" * 40)
    print(f"TO: {person['name']}")
    print("─" * 40)
    email = template.format(
        name=person["name"],
        role=person["role"]
    )
    print(email)
    print()
